// Supabase Edge Function: receives the public enquiry form on customelectrics.co.uk,
// writes it into `leads`, and emails Will an alert.
//
// WHY THIS EXISTS AT ALL, rather than the browser writing to Supabase directly:
// a public form posting straight to PostgREST would need an RLS policy granting
// `anon` INSERT on `leads`. That is a bigger hole than it looks — anon would be
// able to forge business_id, status, lead_cost and source, and any mistake in the
// policy (or a later policy edit) risks exposing the customer list on SELECT.
// Running server-side with the service-role key means `leads` keeps its existing
// member-only policies and the anon key gains NOTHING. The only public surface is
// this function, which decides exactly which columns get written.
//
// DEPLOY (note the flag — without it Supabase demands a signed-in caller and the
// public form gets a 401):
//   supabase functions deploy website-enquiry --no-verify-jwt
//
// Secrets required (Project Settings -> Edge Functions -> Secrets):
//   RESEND_API_KEY        same key the send-email function uses
//   ENQUIRY_BUSINESS_ID   (optional) the businesses.id to file leads against.
//                         If unset, and there is exactly one row in `businesses`,
//                         that one is used. With more than one it fails loudly
//                         rather than guessing and filing leads to the wrong business.
//
// SUPABASE_SERVICE_ROLE_KEY and SUPABASE_URL are injected by the platform.

import { serve } from "https://deno.land/std@0.224.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const FROM_ADDRESS = "will@customelectrics.co.uk";
const ALERT_TO = "will@customelectrics.co.uk";

// Only our own site may post here. A wildcard would let anyone embed the form on
// their own page and pump junk into the leads table from a domain we can't see.
const ALLOWED_ORIGINS = [
  "https://customelectrics.co.uk",
  "https://www.customelectrics.co.uk",
];

function corsHeaders(origin: string | null) {
  const allowed = origin && ALLOWED_ORIGINS.includes(origin) ? origin : ALLOWED_ORIGINS[0];
  return {
    "Access-Control-Allow-Origin": allowed,
    "Access-Control-Allow-Headers": "content-type",
    "Access-Control-Allow-Methods": "POST, OPTIONS",
    "Vary": "Origin",
  };
}

const clean = (v: unknown, max: number): string | null => {
  if (typeof v !== "string") return null;
  const s = v.trim().slice(0, max);
  return s.length ? s : null;
};

serve(async (req) => {
  const origin = req.headers.get("Origin");
  const CORS = corsHeaders(origin);
  const json = (body: unknown, status = 200) =>
    new Response(JSON.stringify(body), {
      status,
      headers: { ...CORS, "Content-Type": "application/json" },
    });

  if (req.method === "OPTIONS") return new Response(null, { headers: CORS });
  if (req.method !== "POST") return json({ error: "Method not allowed" }, 405);

  try {
    const payload = await req.json().catch(() => ({}));

    // Honeypot. Real people never fill a visually hidden field; bots fill every
    // input they find. Return 200 so the bot believes it succeeded and doesn't
    // retry with a different strategy.
    if (clean(payload.website, 200)) {
      console.log("honeypot triggered, discarding");
      return json({ ok: true });
    }

    const name = clean(payload.name, 120);
    const phone = clean(payload.phone, 40);
    const email = clean(payload.email, 160);
    const address = clean(payload.address, 200);
    const postcode = clean(payload.postcode, 20);
    const service = clean(payload.service, 80) ?? "Other";
    const notes = clean(payload.notes, 4000);

    if (!name) return json({ error: "Name is required" }, 400);
    if (!phone && !email) return json({ error: "A phone number or email is required" }, 400);
    if (email && !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) {
      return json({ error: "That email address doesn't look right" }, 400);
    }

    const supabase = createClient(
      Deno.env.get("SUPABASE_URL")!,
      Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
    );

    // Which business do these leads belong to?
    let businessId = Deno.env.get("ENQUIRY_BUSINESS_ID") ?? null;
    if (!businessId) {
      const { data: businesses, error } = await supabase
        .from("businesses").select("id").limit(2);
      if (error) throw error;
      if (!businesses || businesses.length !== 1) {
        // Guessing here would silently file enquiries against the wrong business,
        // where they'd be invisible to the person who needs to see them.
        console.error("Cannot resolve business_id; set ENQUIRY_BUSINESS_ID");
        return json({ error: "Server not configured" }, 500);
      }
      businessId = businesses[0].id;
    }

    // Cheap duplicate guard: same person, same day. Stops a double-tap on a flaky
    // mobile connection creating two identical leads in the chase list.
    const since = new Date(Date.now() - 10 * 60 * 1000).toISOString();
    const { data: recent } = await supabase
      .from("leads").select("id")
      .eq("business_id", businessId)
      .eq("name", name)
      .gte("created_at", since)
      .limit(1);
    if (recent && recent.length) {
      console.log("duplicate enquiry within 10 minutes, ignoring");
      return json({ ok: true, duplicate: true });
    }

    // source/status/lead_cost are set HERE, never taken from the request. A website
    // enquiry costs nothing to acquire, and that zero is what makes the Leads tab's
    // cost-per-win comparison against Checkatrade and Rated People meaningful.
    const { data: lead, error: insertError } = await supabase
      .from("leads")
      .insert({
        business_id: businessId,
        name,
        phone,
        email,
        address,
        postcode,
        source: "Website",
        service,
        status: "new",
        lead_cost: 0,
        notes,
        enquiry_date: new Date().toISOString().slice(0, 10),
      })
      .select("id")
      .single();

    if (insertError) {
      // Log the detail server-side; never hand a database error to a visitor.
      console.error("lead insert failed:", insertError);
      throw insertError;
    }

    // Alert email. Failing to send must NOT fail the request — the lead is already
    // saved, and telling the customer their enquiry failed would be a lie that
    // makes them go elsewhere.
    try {
      const lines = [
        `New website enquiry from ${name}`,
        "",
        phone ? `Phone:    ${phone}` : null,
        email ? `Email:    ${email}` : null,
        address || postcode ? `Address:  ${[address, postcode].filter(Boolean).join(", ")}` : null,
        `Service:  ${service}`,
        "",
        notes ? `Details:\n${notes}` : "(no details given)",
        "",
        "--",
        "This has been saved to your Leads tab as a new lead.",
        "https://customelectrics.co.uk/leads.html",
      ].filter((l) => l !== null);

      const resp = await fetch("https://api.resend.com/emails", {
        method: "POST",
        headers: {
          "Authorization": `Bearer ${Deno.env.get("RESEND_API_KEY")}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          from: FROM_ADDRESS,
          to: ALERT_TO,
          reply_to: email ?? undefined,
          subject: `New enquiry: ${name}${postcode ? ` (${postcode})` : ""} — ${service}`,
          text: lines.join("\n"),
        }),
      });
      if (!resp.ok) console.error("alert email rejected:", await resp.text());
    } catch (mailErr) {
      console.error("alert email threw:", mailErr);
    }

    return json({ ok: true, id: lead.id });
  } catch (err) {
    console.error("enquiry failed:", err);
    return json({ error: "Could not save your enquiry" }, 500);
  }
});
